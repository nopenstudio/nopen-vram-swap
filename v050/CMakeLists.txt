cmake_minimum_required(VERSION 3.20)
project(nopen_vram_v050 C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(NOPEN_OPENCL "Compile real AMD OpenCL backend; never silently fallback" ON)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../v040" "${CMAKE_CURRENT_BINARY_DIR}/v040")
add_library(nopen_vm_policy STATIC src/nopen_vm_policy.c)
target_include_directories(nopen_vm_policy PUBLIC "${CMAKE_CURRENT_LIST_DIR}/include")
target_compile_options(nopen_vm_policy PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(nopen_vm_sensor STATIC src/nopen_vm_sensor.c)
target_include_directories(nopen_vm_sensor PUBLIC "${CMAKE_CURRENT_LIST_DIR}/include")
target_link_libraries(nopen_vm_sensor PRIVATE m)
target_compile_options(nopen_vm_sensor PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(nopen-vm-probe src/nopen_vm_probe.c)
target_link_libraries(nopen-vm-probe PRIVATE nopen_vm_sensor nopen_vm_policy m)
add_executable(nopen-vm-gate src/nopen_vm_gate.c)
target_link_libraries(nopen-vm-gate PRIVATE nopen_vm_sensor nopen_vm_policy m)
target_compile_options(nopen-vm-gate PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_compile_options(nopen-vm-probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(nopen-vm-test-policy tests/test_vm_policy.c)
target_link_libraries(nopen-vm-test-policy PRIVATE nopen_vm_policy)
target_compile_options(nopen-vm-test-policy PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(nopen-vm-gate-integration-test src/nopen_vm_gate.c tests/fake_vm_sensor.c)
target_link_libraries(nopen-vm-gate-integration-test PRIVATE nopen_vm_policy)
target_compile_options(nopen-vm-gate-integration-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(nopen-vm-test-slow-child tests/slow_child.c)
target_compile_options(nopen-vm-test-slow-child PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(nopen_blk_backend STATIC src/nopen_blk_backend.c)
target_include_directories(nopen_blk_backend PUBLIC "${CMAKE_CURRENT_LIST_DIR}/include")
target_link_libraries(nopen_blk_backend PUBLIC nopen_vram_store nopen_page_core Threads::Threads)
target_compile_options(nopen_blk_backend PRIVATE -Wall -Wextra -Wpedantic -Werror)
find_package(Threads REQUIRED)
add_executable(nopen-vm-test-blk tests/test_blk_backend.c)
target_link_libraries(nopen-vm-test-blk PRIVATE nopen_blk_backend)
target_compile_options(nopen-vm-test-blk PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_test(NAME v050_freestanding_vm_policy COMMAND nopen-vm-test-policy)
add_test(NAME v050_readonly_sensor_help COMMAND nopen-vm-probe --help)
add_test(NAME v050_uffd_supervisor_help COMMAND nopen-vm-gate --help)
add_test(NAME v050_uffd_gate_rejects_invalid COMMAND nopen-vm-gate --backend mock --pool-mib 64 --logical-pages 1024 --ram-pages 32 --timeout-seconds 61)
set_tests_properties(v050_freestanding_vm_policy PROPERTIES PASS_REGULAR_EXPRESSION "VM_POLICY_PASS")
set_tests_properties(v050_readonly_sensor_help PROPERTIES PASS_REGULAR_EXPRESSION "read-only sensor")

set_tests_properties(v050_uffd_gate_rejects_invalid PROPERTIES WILL_FAIL TRUE)
add_test(NAME v050_userfaultfd_pressure_guard_integration
  COMMAND bash "${CMAKE_CURRENT_LIST_DIR}/tests/test_vm_gate.sh"
  $<TARGET_FILE:nopen-vm-gate-integration-test>
  $<TARGET_FILE:nopen-vram-demand>
  $<TARGET_FILE:nopen-vm-test-slow-child>)
set_tests_properties(v050_userfaultfd_pressure_guard_integration PROPERTIES
  TIMEOUT 35 PASS_REGULAR_EXPRESSION "VM_GATE_TEST_PASS")

add_test(NAME v050_checked_gpu_block_io_backend COMMAND nopen-vm-test-blk)
set_tests_properties(v050_checked_gpu_block_io_backend PROPERTIES PASS_REGULAR_EXPRESSION "BLK_BACKEND_PASS")
