cmake_minimum_required(VERSION 3.16)
project(nopen_vram_swap VERSION 0.10.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(Vulkan REQUIRED)
add_executable(nopen-vulkan-probe src/nopen_vulkan_probe.c)
target_link_libraries(nopen-vulkan-probe PRIVATE Vulkan::Vulkan)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(nopen-vulkan-probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
install(TARGETS nopen-vulkan-probe RUNTIME DESTINATION bin)
enable_testing()
add_test(NAME probe-help COMMAND nopen-vulkan-probe --help)
