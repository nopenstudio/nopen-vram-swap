cmake_minimum_required(VERSION 3.18)
project(nopen_opencl_benchmark LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(OpenCL REQUIRED)
add_executable(nopen-opencl-bench nopen_opencl_bench.c)
target_compile_options(nopen-opencl-bench PRIVATE -O2 -Wall -Wextra -Wpedantic)
target_link_libraries(nopen-opencl-bench PRIVATE OpenCL::OpenCL)
enable_testing()
add_test(NAME help COMMAND nopen-opencl-bench --help)
add_test(NAME refuse_oversized COMMAND nopen-opencl-bench --mib 4096)
set_tests_properties(refuse_oversized PROPERTIES WILL_FAIL TRUE)
